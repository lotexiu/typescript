  async buscarOrdemServico(idCodOrd: string) {
    const regex = new RegExp('^[0-9]+$');

    if (!regex.test(idCodOrd)) return;

    // Simula o clique após um pequeno atraso para garantir que o DOM esteja pronto
    setTimeout(() => {
      this.tabDescricao.nativeElement.click();
    }, 100);

    await this.abrirOrdemServico();

    if (this.txtReabrirOS == '') {
      return;
    }

    if (idCodOrd == '0') return;

    this.txtReabrirOS = idCodOrd;
    this.apenasVisualizar = false;
    this.habilitaSalvar(true);
    await this.alterarRegistro(idCodOrd);
  }

  async abrirOrdemServico() {
    this.loadAgregado();
    this._cardManagerService.addCard(this.cardName + this.txtReabrirOS);

    this._usuario = Object.assign(new Usuario(), this._usuarioStorage.get());
    this._usuarioAcao = Object.assign(
      new Array(),
      this._usuarioAcaoStorageService.get()
    );
    this.parametro = this._parametroStorageService.get();

    this.localizarSLACategoria();
    this.abaSelecionada = 'cvDescricao' + this.idOrdemServico;

    if (this.reabrirOS && this.txtReabrirOS != '') {
      this.habilitarComboBox();
      await this.reabrirOrdemServico();
    }

    if (this.alterarOS && this.ordemServico != null) {
      this.habilitarComboBox();
      this.alterarOrdemServico();
    }
  }

  async localizarSLACategoria() {
    if (this.formularioProduto.controls['cmbCategoria'].value != '') {
      let filtros = new Array();
      filtros.push(
        new Filtro(
          FiltroConst.PRIORIZACAO_CATEGORIA,
          this.formularioProduto.controls['cmbCategoria'].value
        )
      );
      try {
        let res = await this._priorizacaoService.localizar(filtros);
        if (AmfUtil.validarResponse(res)) {
          let resultado: Array<any> = res.data.body;
          if (resultado.length <= 0) {
            this.btnAlterarPrioridadeDisabled = true;
            this.formularioOs.controls['txtPrioridadeSLA'].setValue('');
            this._alertService.message(
              NotificationConst.TITLE_ATENCAO,
              'Categoria sem SLA definido!'
            );
            return;
          }
          this.btnAlterarPrioridadeDisabled = false;
          if (
            this.codCatAnterior !=
            this.formularioProduto.controls['cmbCategoria'].value
          ) {
            this.limparSLA();
          }

          let cmbSitAtual = this._cmbSitAtualArray.find(
            (cmbSitAtual) =>
              cmbSitAtual.codsit ==
              this.formularioOs.controls['cmbSitAtual'].value
          );

          if (
            this.formularioOs.controls['cmbSitAtual'].value != '' &&
            this.ordemServico.suptec.codsitant !=
              this.formularioOs.controls['cmbSitAtual'].value &&
            cmbSitAtual?.integtrac === 'S'
          ) {
            this.alterouSLA = true;
          }

          resultado.forEach((priorizacao) => {
            if (
              this.ordemServico != null &&
              this.ordemServico.sla == priorizacao.tipo
            ) {
              this.prazoSLA = priorizacao.prazo;
            }
          });
        }
      } catch (error: any) {
        console.error(error);
        this._notificationService.error(
          NotificationConst.TITLE_DEFAULT,
          error.message
        );
      }
    }
  }

  private limparSLA() {
    this.formularioOs.controls['txtPrioridadeSLA'].setValue('');
    this.slaAnterior = '';
    this.ordemServico.sla = '';
    this.alterouSLA = true;
  }

  private async reabrirOrdemServico() {
    this.alterarLayoutTituloOrdemServico();
    await this.buscarTicketFechado();
  }

  private async buscarTicketFechado() {
    try {
      let res = await this._ordemservicoService.localizar(
        Number(this.txtReabrirOS)
      );

      if (AmfUtil.validarResponse(res)) {
        this.ordemServico = res.data.body;
        if (this.ordemServico.suptec.codticket != null) {
          await this.localizarTicket(this.ordemServico.suptec.codticket);
        }

        let codsit = this.ordemServico.suptec.codsit;
        let situacaoGeraOutroProcedimento: boolean =
          codsit != 1 && codsit != 3 && codsit != 10;

        if (situacaoGeraOutroProcedimento) {
          this._alertService.message(
            NotificationConst.TITLE_ATENCAO,
            'Esta Ordem de Serviço não está finalizada!'
          );
          this.txtReabrirOSDisabled = false;
          this.txtReabrirOS = '';
          return;
        }

        if (!this.reabrirOrdemServicoAlterarOutros) {
          if (this.ordemServico.codigoTecnico != this._usuario.tabven.codigo) {
            this._alertService.message(
              NotificationConst.TITLE_ATENCAO,
              'Você não possui permissão para reabrir O.S. de outro usuário!'
            );
            this.txtReabrirOSDisabled = false;
            this.txtReabrirOS = '';
            return;
          }
        }

        await this.listarAgregadoAposCarregarOS(
          this.ordemServico.suptec.codord,
          this.ordemServico.cliente,
          this.ordemServico.tipoPlanejadoTeste
        );

        let filtros = new Array();
        filtros.push(
          new Filtro(
            FiltroConst.TICKET_HISTORICO_CODIGO_ORDEM_SERVICO,
            this.ordemServico.suptec.codord.toString()
          )
        );

        try {
          let res = await this._ticketHistoricoService.listar(filtros);
          if (AmfUtil.validarResponse(res)) {
            let historicos = res.data.body;

            if (historicos != null && historicos.length > 0) {
              this.ordemServico.ultimoHistorico =
                historicos[historicos.length - 1];
            } else {
              this.ordemServico.ultimoHistorico = new TicketHistorico();
            }

            this.apenasVisualizar = false;
            await this.carregarDadosCampos().then(() => {
              this.formularioOs.controls['cmbSitAtual'].setValue('');
            });
            this.carregarTitulo();
          }
        } catch (error: any) {
          console.error(error);
          this._notificationService.error(
            NotificationConst.TITLE_DEFAULT,
            error.message
          );
        }
        this._alertService.questionYesNo(
          NotificationConst.TITLE_ATENCAO,
          'Deseja reabrir a O.S. #' + this.ordemServico.suptec.codord + '?',
          () => this.confirmacaoParaReabrirOS(),
          () => this
        );
      } else {
        this._alertService.message(
          NotificationConst.TITLE_ATENCAO,
          'Esta Ordem de Serviço não existe!'
        );
        this.txtReabrirOSDisabled = false;
        this.carregarFormulario();
        this.txtReabrirOS = '';
      }
    } catch (error: any) {
      console.error(error);
      this._notificationService.error(
        NotificationConst.TITLE_DEFAULT,
        error.message
      );
    }
  }

  private async listaModulos() {

        FiltroConst.PRODUTO_MODULO_CODIGO_PRODUTO,
        this.formularioProduto.controls['cmbProduto'].value

    );

    try {
      let res = await this._moduloService.listar(filtro);

      if (AmfUtil.validarResponse(res)) {
        this._cmbModulos = res.data.body.map(
          (modulo: any) =>
            new Option(`${modulo.codigo} - ${modulo.descricao}`, modulo.codigo)
        );

        let modulos = res.data.body.find(
          (modulo: any) => modulo.codigo == this.ordemServico.suptec.modulatend
        );

          this.formularioProduto.controls['cmbModulos'].setValue(
            this.ordemServico.suptec.modulatend.toString()

          this.formularioProduto.controls['cmbModulos'].setValue('');

      }
    } catch (error: any) {

      this._notificationService.error(
        NotificationConst.TITLE_DEFAULT,

      );

  async alterarRegistro(codord: string) {
    try {
      this.carregando = true;

      let res = await this._ordemservicoService.localizar(Number(codord));
      if (AmfUtil.validarResponse(res)) {
        this.ordemServico = Object.assign(new OrdemServico(), res.data.body);

        if (this.ordemServico.suptec.codsit == 1) this.apenasVisualizar = true;

        this.codCatAnterior = this.ordemServico.suptec.codcat;
        this.habilitarHBoxRetorno();
        this.habilitarGerarSenhaSuporte();
        this.btnHBoxTicket =
          this.ordemServico.suptec.codticket != null &&
          this.ordemServicoAlterar;
        this.btnHBoxReleaseNotes =
          this.ordemServico.suptec.codticket != null &&
          this.ordemServicoReleaseNotesAlterar;

          if (this.ordemServico.suptec.codticket != null) {
            await this.localizarTicket(
            this.ordemServico.suptec.codticket.toString()
          );
        }

        this.situacaoGeraOutroProcedimento =
          this.ordemServico.suptec.outroProc == 'S';

        if (this.apenasVisualizar) {
          if (
            this.ordemServico.codigoTecnico == this._usuario.fkIdVendedor &&
            this.ordemServicoAlterar
          ) {
            this.visualizarHBoxEditar =
              this.apenasVisualizar && this.situacaoGeraOutroProcedimento;
          } else if (
            this.ordemServico.codigoTecnico != this._usuario.fkIdVendedor &&
            this.ordemServicoAlterarOutros
          ) {
            this.visualizarHBoxEditar =
              this.apenasVisualizar && this.situacaoGeraOutroProcedimento;
          } else {
            this.visualizarHBoxEditar = false;
          }
        } else {
          this.visualizarHBoxEditar = false;
          if (
            this.ordemServico.codigoTecnico == this._usuario.fkIdVendedor &&
            this.ordemServicoAlterar
          ) {
            this.apenasVisualizar = false;
          } else if (
            this.ordemServico.codigoTecnico != this._usuario.fkIdVendedor &&
            this.ordemServicoAlterarOutros
          ) {
            this.apenasVisualizar = false;
          } else {
            this.apenasVisualizar = true;
          }
        }
        this.formularioOs.controls['txtPrioridadeSLA'].setValue(
          this.ordemServico.sla
        );
        this.slaAnterior = this.ordemServico.sla;
        this.formularioOs.controls['txtClassificacao'].setValue(
          this.ordemServico.classificacao
        );

        this.motivoResultadoAlteracaoOS = new MotivoResultadoAlteracaoOS();
        this.motivoResultadoAlteracaoOS.fkIdOrdemServico =
          this.ordemServico.suptec.codord;

        this.carregando = false;

        await this.listarAgregadoAposCarregarOS(
          this.ordemServico.suptec.codord,
          this.ordemServico.cliente,
          this.ordemServico.tipoPlanejadoTeste
        );
        this.listarNotificacoes();
        this.habilitarCampos();
        if (this.reabrirOS) {
          this.formularioOs.controls['cmbSitAtual'].setValue('');
          this.apenasVisualizar = false;
          this.habilitarComboBox();
        }
      } else {
        await this._alertService.message(
          NotificationConst.TITLE_ATENCAO,
          'Ordem de servico não cadastrada!'
        );
        this.carregando = false;
        this.desabilitarComboBox();
        this.desabilitarCampos();
        this.desabilitarBotoes();
        this.limparTela();
      }
      this.listarAnexos();
    } catch (error: any) {
      console.error(error);
      this._notificationService.error(
        NotificationConst.TITLE_DEFAULT,
        error.message
      );
    }
  }

  private async listarAgregadoAposCarregarOS(
    codord: number,
    cliente: Cliente,
    tipoPlanejadoTeste: String
  ) {
    try {
      let res = await this._ordemservicoService.listarAgregadoAposCarregarOS(
        codord,
        cliente,
        tipoPlanejadoTeste
      );
      if (AmfUtil.validarResponse(res)) {
        let resultado = Object.assign(new Array(), res.data.body);

        /*ultimoTicketHistorico*/
        let resultadoUltimoTicketHistorico: TicketHistorico =
          resultado[0] as TicketHistorico;

        await this.onResultBuscarTicketHistorico(
          resultadoUltimoTicketHistorico
        );

        this.resultadoTemAtendimento = resultado[5];

        /*vincularOS*/
        let resultadoVincularOS = resultado[1];
        this.btnVincularOS =
          'C ' +
          resultadoVincularOS.qtdeConsulta.toString() +
          '  D ' +
          resultadoVincularOS.qtdeDependente.toString();

        /*ticketTestado*/
        let resultadoTicketTestado = resultado[2];
        if (this.ordemServico.tipoPlanejadoTeste == 'ST') {
          this.mostrarIconeTeste(OrdemServicoComponent.NAO_PRECISA_TESTAR);
        } else if (resultadoTicketTestado.length == 0) {
          this.mostrarIconeTeste(OrdemServicoComponent.NAO_TESTADO);
        } else {
          this.mostrarIconeTeste(OrdemServicoComponent.TESTADO);
        }

        /*verificarRevisao*/
        let resultadoVerificarRevisao = resultado[3];
        this.ticketPossuiRevisao = resultadoVerificarRevisao > 0;

        /*motivoResultadoAlteracaoOS*/
        let resultadoMotivoResultadoAlteracaoOS: MotivoResultadoAlteracaoOS =
          resultado[4];
        if (resultadoMotivoResultadoAlteracaoOS.id > 0) {
          this.motivoResultadoAlteracaoOS = resultadoMotivoResultadoAlteracaoOS;
        }
      }
      this.mostrarIconeMotivoResultadoAlteracaoOS();

      this.carregarTitulo();
    } catch (error: any) {
      console.error(error);
      this._notificationService.error(
        NotificationConst.TITLE_DEFAULT,
        error.message
      );
    }
  }

  private carregarTipoAtendimento() {
    if (
      this.ordemServico.ultimoHistorico.tipoAtendimento !=
        TipoAtendimento.VALOR_FINALIZADO &&
      this.ordemServico.ultimoHistorico.id > 0 &&
      this.ordemServico.ultimoHistorico.dataFim == null
    ) {
      this.formularioOs.controls['cmbTipoAtendimento'].setValue(
        this.ordemServico.ultimoHistorico.tipoAtendimento
      );
      let tipoAtendimento =
        this._tipoDeAtendimento.find(
          (atendimento) =>
            atendimento.value ==
            this.formularioOs.controls['cmbTipoAtendimento'].value
        ) || new TipoDeAtendimento();
      this.historioAtendimento = tipoAtendimento.label;
    } else if (this.ordemServico.ultimoHistorico.id == 0) {
      this.formularioOs.controls['cmbTipoAtendimento'].setValue(
        TipoAtendimento.VALOR_NENHUM
      );
      this.historioAtendimento = TipoAtendimento.LABEL_NENHUM;
    } else {
      this.formularioOs.controls['cmbTipoAtendimento'].setValue(
        TipoAtendimento.VALOR_FINALIZADO
      );
      this.historioAtendimento = TipoAtendimento.LABEL_FINALIZADO;
    }
  }

  private carregarTitulo() {
    let usuarioAberturaOS: string =
      this.ordemServico.suptec.usuaber != ''
        ? ' - ' + this.ordemServico.suptec.usuaber
        : '';
    this.msgFinalizado = false;
    let msgRetorno: Boolean = false;
    this.msgAtendiementoAberto = false;
    let linkButtonText: String = '';
    let tituloLink = '';

    this.titleText = usuarioAberturaOS;
    let titulo = this.titleText.toString();

    let tituloContinuaDa = '';
    let linkButtonTextContinuaDa = '';
    let codOrdContinuaDa = '';
    let tituloContinuaNa = '';
    let linkButtonTextContinuaNa = '';
    let codOrdContinuaNa = '';

    let linkButtonTextRetornoDa = '';
    let codOrdRetornoDa = '';
    let tituloRetornoDa = '';
    let linkButtonTextOrigemRetorno = '';
    let codOrdOrigemRetorno = '';
    let tituloOrigemretorno = '';
    let linkAtendimentoAberto = '';

    this.isContinuaNa = false;
    this.isContinuaDa = false;

    this.tituloAtendimentoAberto = false;
    this.tituloButtonOrigenRetorno = false;
    this.tituloButtonRetorno = false;
    this.tituloButtonContDa = false;
    this.tituloButtonContNa = false;

    //Tratamento Finalizado
    if (this.ordemServico.suptec.usufinal != null) {
      if (this.ordemServico.suptec.usufinal.toString() != '') {
        let dataFim: string =
          DateField.dateToString(
            new Date(this.ordemServico.protec.datafim),
            'DD/MM/YYYY'
          ) || '';
        this.titleText =
          this.titleText +
          ' - Finalizada ' +
          this.ordemServico.suptec.usufinal +
          ' ' +
          dataFim +
          ' ' +
          this.ordemServico.protec.horafim;
        this.msgFinalizado = true;
        titulo = this.titleText.toString();
      }
    } else {
      if (this.resultadoTemAtendimento) {
        this.msgAtendiementoAberto = true;
        this.tituloAtendimentoAberto = true;
        linkAtendimentoAberto += `<button style="border: none; background-color: transparent; color: #CCC000"> - Atendimento aberto</button>`;

        titulo = this.titleText.toString();
      }
    }
    this.titleText = '';

    //msgRetorno
    if (this.ordemServico.suptec.codordorig != null) {
      tituloRetornoDa =
        this.titleText +
        `<label style="height: 0px; color: #000000"> - Retorno da <label>`;
      linkButtonText = '';
      this.tituloButtonRetorno = true;
      codOrdRetornoDa = this.ordemServico.suptec.codordorig
        ? this.ordemServico.suptec.codordorig.toString()
        : '';
      linkButtonTextRetornoDa =
        `<button style="border: none; background-color: transparent; font-weight:bold;">` +
        codOrdRetornoDa +
        `</button>`;

      msgRetorno = true;
      titulo = titulo + this.titleText.toString();
      this.titleText = '';
    }

    //msgOriginouRetorno
    if (this.ordemServico.retorno != 'N') {
      tituloOrigemretorno =
        this.titleText +
        `<label style="height: 0px; color: #000000"> - Originou Retorno <label>`;
      linkButtonText = '';
      this.tituloButtonOrigenRetorno = true;
      codOrdOrigemRetorno = this.ordemServico.retorno;
      msgRetorno = true;

      linkButtonTextOrigemRetorno =
        `<button style="border: none; background-color: transparent; font-weight:bold;">` +
        codOrdOrigemRetorno +
        `</button>`;
      titulo = titulo + this.titleText.toString();
      this.titleText = '';
    }

    //msgOriginouContinuação
    if (this.ordemServico.suptec.codordcont > 0) {
      linkButtonText = '';
      this.titleText = this.titleText + ' - Continuação da ';
      this.tituloButtonContDa = true;
      codOrdContinuaDa = this.ordemServico.suptec.codordcont
        ? this.ordemServico.suptec.codordcont.toString()
        : '';
      tituloContinuaDa = `<label style="height: 0px;"> - Continuação da </label>`;
      linkButtonTextContinuaDa =
        `<button style="border: none; background-color: transparent; font-weight:bold;">` +
        codOrdContinuaDa +
        `</button>`;

      this.titleText = '';
      msgRetorno = true;
      this.isContinuaDa = true;
    }

    //msgContinuação
    if (this.ordemServico.continuacao != 'N') {
      linkButtonText = '';
      this.titleText = this.titleText + ' - Continua na ';
      linkButtonText = this.ordemServico.continuacao;
      msgRetorno = true;

      tituloLink += `<label style="height: 0px;" >${this.titleText.toString()}<label>`;
      this.titleText = '';
      this.tituloButtonContNa = true;
      codOrdContinuaNa = this.ordemServico.continuacao;
      tituloContinuaNa = `<label style="height: 0px;"> - Continua na </label>`;
      linkButtonTextContinuaNa =
        `<button style="border: none; background-color: transparent; font-weight:bold;">` +
        codOrdContinuaNa +
        `</button>`;

      this.isContinuaNa = true;
    }

    this.tituloButtonContDaInnerHTML.nativeElement.innerHTML =
      tituloContinuaDa + ' ' + linkButtonTextContinuaDa;
    this.tituloButtonContNaInnerHTML.nativeElement.innerHTML =
      tituloContinuaNa + ' ' + linkButtonTextContinuaNa;
    this.tituloButtonRetornoDaInnerHTML.nativeElement.innerHTML =
      tituloRetornoDa + ' ' + linkButtonTextRetornoDa;
    this.tituloButtonOrigenRetornoInnerHTML.nativeElement.innerHTML =
      tituloOrigemretorno + ' ' + linkButtonTextOrigemRetorno;
    this.tituloAtendimentoAbertoInnerHTML.nativeElement.innerHTML =
      linkAtendimentoAberto;

    this.tituloInnerHTML.nativeElement.innerHTML = `<label style="height: 0px;">${titulo}<label>`;

    this.titulo = titulo;
    this.verificarPrazo(this.ordemServico.suptec.codord.toString());
  }

  private async verificarPrazo(codord: string) {
    let filtro = new Array();

    filtro.push(new Filtro(FiltroConst.ORDPREV_CODORD, codord));
    filtro.push(new Filtro(FiltroConst.TIPO_PRAZO_ENTREGA_CLIENTE, ''));

    this.prazoOS = '';

    try {
      let res = await this._ordprevService.listar(filtro);

      if (AmfUtil.validarResponse(res)) {
        this._gdPrevisao = res.data.body;

        this._gdPrevisao.forEach((ordprev) => {
          if (ordprev.tipoPrazoPrevisao.nome == 'C') {
            this.prazoOS =

      );

      let res = await this._treinamentoService.listar(filtro);

      }
    } catch (error: any) {

  async cmbSitAtual_change() {
    if (this.formularioOs.controls['cmbSitAtual'].value != '') {
      let situacao = this._situacaoAtual.find(
        (situacao) =>
          situacao.codsit == this.formularioOs.controls['cmbSitAtual'].value
      );
      let parceiroSituacaoPassarOSSG: Parametro =
        this.parametro.find(
          (parametro) =>
            parametro.nomeParametro ==
            ParametroConst.PARAMETRO_PARCEIRO_SITUACAO_PASSAR_OS_SG
        ) || new Parametro();

      if (!situacao) {
        return;
      }

      if (situacao.outro_proc && situacao.outro_proc == 'S') {
        this.txtSolucaoDisabled = true;
        this.formularioOs.controls['cmbTipoAtendimento'].enable();
      } else if (
        situacao.outro_proc == null &&
        this.formularioOs.controls['cmbTipoAtendimento'].value ==
          TipoAtendimento.VALOR_NENHUM
      ) {
        this.txtSolucaoDisabled = this.apenasVisualizar;
        this.formularioOs.controls['cmbTipoAtendimento'].disable();
      } else if (
        this.formularioOs.controls['cmbTipoAtendimento'].value !=
        TipoAtendimento.VALOR_NENHUM
      ) {
        this.txtSolucaoDisabled = this.apenasVisualizar;
        this.formularioOs.controls['cmbTipoAtendimento'].setValue(
          TipoAtendimento.VALOR_FINALIZADO
        );
        this.formularioOs.controls['cmbTipoAtendimento'].disable();
      } else {
        this.txtSolucaoDisabled = this.apenasVisualizar;
        !this.apenasVisualizar
          ? this.formularioOs.controls['cmbTipoAtendimento'].disable()
          : this.formularioOs.controls['cmbTipoAtendimento'].enable();
      }

      if (situacao.codsit != this.ordemServico.suptec.codsitant) {
        this.ordemServico.abrirOSParaSG = false;

        if (
          this.ordemServico.fkIdOrdemServicoSG <= 0 &&
          situacao.codsit == parceiroSituacaoPassarOSSG.valorParametro
        ) {
          await this._alertService.questionYesNo(
            NotificationConst.TITLE_ATENCAO,
            'Deseja encaminhar Ordem de Serviço para SG?',
            () => {
              this.ordemServico.abrirOSParaSG = true;
            },
            () => this
          );
        }
      }
    }

  private validarAgregado(
    automatico: boolean,
    eventoF3: boolean,
    fecharTelaOrdemServico: boolean
  ): Boolean {
    if (!this.validarComboboxes()) {
      return false;
    }
    if (!this.validarComboboxCategoria()) {
      return false;
    }
    if (!this.validarSolucao(automatico)) {
      return false;
    }
    if (!this.validarFinalizarOSEmpresaParceira()) {
      return false;
    }
    if (!this.validarAcaoAlterarCategoria()) {
      return false;
    }
    if (!this.validarAcaoTransferirCategoria()) {
      return false;
    }
    if (!this.validarAcaoFinalizarCategoria()) {
      return false;
    }
    if (!this.validarAcaoAlterarOSCategoria()) {
      return false;
    }
    if (!this.validarAcaoAlterarProduto()) {
      return false;
    }
    if (!this.validarAcaoPassarOS()) {
      return false;
    }
    if (!this.validarAcaoFinalizarOS()) {
      return false;
    }
    if (!this.validarAcaoMarcarTestadoOS()) {
      return false;
    }
    if (!this.validarFinalizacaoOSAtendimentoEmAberto()) {
      return false;
    }
    if (!this.validarCampoCliente()) {
      return false;
    }
    if (!this.validarCampoContato()) {
      return false;
    }
    if (!this.validarSituacaoReaberturaOS()) {
      return false;
    }
    if (!this.validarSla(automatico, fecharTelaOrdemServico)) {
      return false;
    }
    if (!this.validarOrdemServicoTestada(automatico, fecharTelaOrdemServico)) {
      return false;
    }
    if (!this.validarSituacaoOSSubirVersao(automatico)) {
      return false;
    }
    if (!this.validarSituacaoOSInformacoesTrac(automatico)) {
      return false;
    }

    return true;
  }

  private validarComboboxes(): boolean {
    if (!this.validarCampos('arrayValidacoes')) {
      this.habilitaSalvar(true);
      return false;
    }

    return true;
  }

  private validarComboboxCategoria(): boolean {
    if (!this.validarCampos('arrayValidacoesCategoria')) {
      this.habilitaSalvar(true);
      return false;
    }

    return true;
  }

  private validarSolucao(automatico: boolean): boolean {
    if (!this.validarCampos('arrayValidacoes')) {
      this.habilitaSalvar(true);
      return false;

    }

    let situacao = this._situacaoAtual.find(
      (situacao) =>
        situacao.codsit == this.formularioOs.controls['cmbSitAtual'].value
    );
    if (automatico && !situacao.outro_proc) {
      return false;
    }

    if (situacao.outro_proc == null) {
      if (!this.validarCampos('regExpTxtSolucao')) {
        this.habilitaSalvar(true);
        return false;
      }
    }

    return true;
  }

  private validarFinalizarOSEmpresaParceira(): boolean {
    if (
      this.ordemServico.fkIdOrdemServicoSG > 0 &&
      !this.usuarioTemAcaoFinalizarOSEmpresaParceira(
        UsuarioAcaoConst.FINALIZAR_OS_VINCULADA_EMPRESA_PARCEIRA
      ) &&
      this.formularioOs.controls['cmbSitAtual'].value == 1
    ) {
      this._notificationService.warning(
        NotificationConst.TITLE_WARNING,
        'Sem permissão para finalizar O.S. vinculada a empresa parceira'
      );
      this.habilitaSalvar(true);
      return false;
    }
    return true;
  }

  private usuarioTemAcaoFinalizarOSEmpresaParceira(acao: String): boolean {
    if (!this.temAcaoFinalizarOSEmpresaParceira(acao)) {
      return true;
    }

    let permitirFinalizarOSEmpresaParceira: boolean = false;

    for (const usuarioAcao of this._usuarioAcao) {
      if (
        usuarioAcao.fkIdUsuario == this._usuario.id &&
        usuarioAcao.acao == acao
      ) {
        return true;
      }
    }

    return permitirFinalizarOSEmpresaParceira;
  }

  private temAcaoFinalizarOSEmpresaParceira(acao: String): boolean {
    for (const usuarioAcao of this._usuarioAcao) {
      if (usuarioAcao.acao == acao) {
        return true;
      }
    }
    return false;
  }

  private async validarAcaoAlterarCategoria() {
    if (!this.usuarioTemAcaoAlterarCategoria()) {
      this._notificationService.warning(
        NotificationConst.TITLE_WARNING,
        'Sem permissão para alterar esta categoria!'
      );
      this.formularioProduto.controls['cmbCategoria'].setValue(
        this.codCatAnterior
      );
      this.habilitaSalvar(true);
      return false;
    }

    return true;
  }

  private usuarioTemAcaoAlterarCategoria(): boolean {
    let permitirAlterarCategoria: boolean = true;

    if (
      this.codCatAnterior ==
      this.formularioProduto.controls['cmbCategoria'].value
    ) {
      return permitirAlterarCategoria;
    }

    for (const usuarioAcao of this._usuarioAcao) {
      if (
        usuarioAcao.acao == UsuarioAcaoConst.ALTERAR_OS_CATEGORIA &&
        usuarioAcao.complementoAcao == this.codCatAnterior
      ) {
        permitirAlterarCategoria = false;
        if (usuarioAcao.fkIdUsuario == this._usuario.id) {
          return true;
        }
      }
    }
    return permitirAlterarCategoria;
  }

  private validarAcaoTransferirCategoria(): boolean {
    if (!this.usuarioTemAcaoTransferirCategoria()) {
      this._notificationService.warning(
        NotificationConst.TITLE_WARNING,
        'Sem permissão para transferir esta categoria!'
      );
      this.formularioProduto.controls['cmbCategoria'].setValue(
        this.codCatAnterior
      );
      this.habilitaSalvar(true);
      return false;
    }

    return true;
  }

  private usuarioTemAcaoTransferirCategoria(): boolean {
    let permitirTransferirCategoria: boolean = true;

    if (
      this.formularioProduto.controls['cmbTecnicos'].value ==
      this.ordemServico.codigoTecnico
    ) {
      return permitirTransferirCategoria;
    }

    for (const usuarioAcao of this._usuarioAcao) {
      if (
        usuarioAcao.acao == UsuarioAcaoConst.TRANSFERIR_OS_CATEGORIA &&
        usuarioAcao.complementoAcao ==
          this.formularioProduto.controls['cmbCategoria'].value
      ) {
        permitirTransferirCategoria = false;
        if (usuarioAcao.fkIdUsuario == this._usuario.id) {
          return true;
        }
      }
    }

    return permitirTransferirCategoria;
  }

  private validarAcaoFinalizarCategoria(): boolean {
    if (!this.usuarioTemAcaoFinalizarCategoria()) {
      this._notificationService.warning(
        NotificationConst.TITLE_WARNING,
        'Sem permissão para finalizar esta categoria!'
      );

      return false;

  }

  private usuarioTemAcaoFinalizarCategoria(): boolean {

    if (this.formularioOs.controls['cmbSitAtual'].value != 1) {
      return permitirFinalizarCategoria;
    }

      if (
        usuarioAcao.acao ==
          UsuarioAcaoConst.FINALIZAR_TRANSFERIR_OS_CATEGORIA &&
        usuarioAcao.complementoAcao ==
          this.formularioProduto.controls['cmbCategoria'].value

        if (usuarioAcao.fkIdUsuario == this._usuario.id) {
          return true;

  }

  private usuarioTemAcaoAlterarOSCategoria(): boolean {

      return permitirAlterarCategoria;
    }

    for (const usuarioAcao of this._usuarioAcao) {
      if (
        usuarioAcao.acao == UsuarioAcaoConst.ALTERAR_OS_CATEGORIA &&

    ) {

      );
      this.formularioProduto.controls['cmbProduto'].setValue(

      );

      return false;
    }

  }

  private usuarioTemAcaoAlterarProduto(acao: String): boolean {
    let permitirAlterarProduto: boolean = true;

    if (
      this.ordemServico.suptec.codprodsg !=
      this.formularioProduto.controls['cmbProduto'].value
    ) {
      for (const usuarioAcao of this._usuarioAcao) {
        if (
          usuarioAcao.acao == acao &&
          usuarioAcao.complementoAcao == this.ordemServico.suptec.codprodsg
        ) {
          permitirAlterarProduto = false;
          if (usuarioAcao.fkIdUsuario == this._usuario.id) {
            return true;
          }
        }
      }
    }

    return permitirAlterarProduto;
  }

  private validarAcaoPassarOS(): boolean {
    if (
      this.ordemServico.codigoTecnico !=
      this.formularioProduto.controls['cmbTecnicos'].value
    ) {
      if (
        !this.verificarAcaoUsuarioComplemento(
          this.formularioProduto.controls['cmbTecnicos'].value,
          UsuarioAcaoConst.USUARIO_PASSAR_OS
        )
      ) {
        this._notificationService.warning(
          NotificationConst.TITLE_WARNING,
          'Sem permissão de transferir O.S para usuario'
        );
        this.formularioProduto.controls['cmbCategoria'].setValue(
          this.codCatAnterior
        );
        this.habilitaSalvar(true);
        return false;
      }
    }

    return true;
  }

  private verificarAcaoUsuarioComplemento(
    tecnico: number,
    acaoUsuario: String
  ): boolean {
    if (tecnico == this._usuario.fkIdVendedor) {
      return true;
    }

    let permitirAcao: boolean = true;
    for (const usuarioAcao of this._usuarioAcao) {
      if (
        usuarioAcao.acao == acaoUsuario &&
        usuarioAcao.complementoAcao == tecnico
      ) {
        permitirAcao = false;
        if (usuarioAcao.fkIdUsuario == this._usuario.id) {
          return true;
        }
      }
    }

    return permitirAcao;
  }

  private validarAcaoFinalizarOS(): boolean {
    let situacao = this._situacaoAtual.find(
      (situacao) =>
        situacao.codsit == this.formularioOs.controls['cmbSitAtual'].value
    );

    if (
      this.ordemServico.codigoTecnico !=
        this.formularioProduto.controls['cmbTecnicos'].value ||
      (this.formularioOs.controls['cmbSitAtual'].value != '' &&
        this.formularioOs.controls['cmbSitAtual'].value !=
          this.ordemServico.suptec.codsitant &&
        situacao.outro_proc != 'S')
    ) {
      if (
        !this.verificarAcaoUsuarioComplemento(
          this.ordemServico.codigoTecnico,
          UsuarioAcaoConst.USUARIO_FINALIZAR_TRANSFERIR_OS
        )
      ) {
        this._notificationService.warning(
          NotificationConst.TITLE_WARNING,
          'Sem permissão de Finalizar/Transferir O.S do usuario!'
        );
        this.habilitaSalvar(true);
        return false;
      }
    }
    return true;
  }

  private validarAcaoMarcarTestadoOS(): boolean {
    let tecnico = this._tecnicos.find(
      (tecnico) =>
        tecnico.codigo == this.formularioProduto.controls['cmbTecnicos'].value
    );

    if (
      this.ordemServico.codigoTecnico !=
      this.formularioProduto.controls['cmbTecnicos'].value
    ) {
      if (
        !this.usuarioTemAcaoTransferirOsTestada(
          this.formularioProduto.controls['cmbTecnicos'].value,
          UsuarioAcaoConst.USUARIO_OS_MARCADA_COMO_TESTADO,
          this.ordemServicoTestada,
          this.ticketPossuiRevisao
        )
      ) {
        this._notificationService.warning(
          NotificationConst.TITLE_WARNING,
          'Obrigatório marcar O.S como testado para transferir para o usuário ' +
            tecnico.nome
        );
        this.habilitaSalvar(true);
        return false;
      }
    }
    return true;
  }

  private usuarioTemAcaoTransferirOsTestada(
    tecnico: number,
    acao: String,
    ordemServicoTestada: boolean,
    ticketPossuiRevisao: boolean
  ): boolean {
    var permitirAlterarUsuario: boolean = true;

    for (const usuarioAcao of this._usuarioAcao) {
      if (
        usuarioAcao.acao == acao &&
        usuarioAcao.fkIdUsuario == tecnico &&
        !ordemServicoTestada &&
        ticketPossuiRevisao
      ) {
        return false;
      }
    }
    return permitirAlterarUsuario;
  }

  private validarFinalizacaoOSAtendimentoEmAberto(): boolean {
    let situacao = this._situacaoAtual.find(
      (situacao) =>
        situacao.codsit == this.formularioOs.controls['cmbSitAtual'].value
    );

    if (
      situacao.outro_proc != 'S' &&
      this.formularioOs.controls['cmbTipoAtendimento'].value != '' &&
      this.formularioOs.controls['cmbTipoAtendimento'].value !=
        TipoAtendimento.VALOR_FINALIZADO &&
      this.formularioOs.controls['cmbTipoAtendimento'].value !=
        TipoAtendimento.VALOR_NENHUM
    ) {
      this._notificationService.warning(
        NotificationConst.TITLE_WARNING,
        'Não é possivel finalizar O.S. com Atendimento Aberto.'
      );
      return false;
    }

    return true;
  }

  private validarCampoCliente(): boolean {
    if (this.ordemDeServicoAlterarCliente) {
      if (
        this.formularioCliente.controls['txtCliente'].value == '' ||
        this.formularioCliente.controls['txtCnpj'].value == ''
      ) {
        this._notificationService.warning(
          NotificationConst.TITLE_WARNING,
          'Preencha o cliente da O.S.!'
        );
        this.txtClienteHTML.nativeElement.focus();
        this.habilitaSalvar(true);
        return false;
      }
      this.ordemServico.suptec.codcli =
        this.ordemServico.cliente.cadcli.codcli10;
    }
    return true;
  }

  private validarCampoContato(): boolean {
    if (this.ordemDeServicoContatoAlterar) {
      if (this.formularioCliente.controls['txtContato'].value == '') {
        this._notificationService.warning(
          NotificationConst.TITLE_WARNING,
          'Preencha o contato da O.S.'
        );
        this.txtContatoHTML.nativeElement.focus();
        this.habilitaSalvar(true);
        return false;
      }
      this.ordemServico.suptec.contato =
        this.formularioCliente.controls['txtContato'].value;
    }
    return true;
  }

  private validarSituacaoReaberturaOS(): boolean {
    let situacao = this._situacaoAtual.find(
      (situacao) =>
        situacao.codsit == this.formularioOs.controls['cmbSitAtual'].value
    );

    if (situacao.outro_proc != 'S') {
      if (this.reabrirOS) {
        this._notificationService.warning(
          NotificationConst.TITLE_WARNING,
          'O.S. já finalizada! Selecione outra situação para reabri-la!'
        );
        this.habilitaSalvar(true);
        return false;
      }
    }
    return true;
  }

  private validarSla(
    automatico: boolean,
    fecharTelaOrdemServico: boolean
  ): boolean {
    let situacao = this._situacaoAtual.find(
      (situacao) =>
        situacao.codsit == this.formularioOs.controls['cmbSitAtual'].value
    );

    if (
      this.formularioOs.controls['cmbSitAtual'].value != '' &&
      situacao.outro_proc != 'S' &&
      !this.slaValido(automatico)
    ) {
      this.habilitaSalvar(true);
      this.abrirTelaSlaOS(automatico, fecharTelaOrdemServico);
      return false;
    }

    if (
      !automatico &&
      this.usuarioTemAcaoSLAObrigatorio() &&
      !this.slaValido(automatico)
    ) {
      this.habilitaSalvar(true);
      this.abrirTelaSlaOS(automatico, fecharTelaOrdemServico);
      return false;
    }

    return true;
  }

  private slaValido(automatico: boolean): boolean {
    return (
      automatico || this.formularioOs.controls['txtPrioridadeSLA'].value != ''
    );
  }

  private usuarioTemAcaoSLAObrigatorio(): boolean {
    let temAcaoSLAObrigatorio: boolean = false;
    let tecnico = this._tecnicos.find(
      (tecnico) =>
        tecnico.codigo == this.formularioProduto.controls['cmbTecnicos'].value
    );

    for (const usuarioAcao of this._usuarioAcao) {
      if (
        usuarioAcao.acao == UsuarioAcaoConst.SLA_OBRIGATORIO &&
        usuarioAcao.complementoAcao == tecnico.fkIdUsuario
      ) {
        return false;
      }
    }

    return temAcaoSLAObrigatorio;
  }

  private validarTecnicoGravacaoAutomatica(automatico: boolean): boolean {
    if (
      automatico &&
      this.formularioProduto.controls['cmbTecnicos'].value !=
        this.ordemServico.codigoTecnico
    ) {
      this.habilitaSalvar(true);
      return false;
    }
    return true;
  }

  private validarOrdemServicoTestada(
    automatico: boolean,
    fecharTelaOrdemServico: boolean
  ): boolean {
    if (!this.ordemServicoTestadoValido(false, automatico)) {
      this.abrirTelaTestado(false, automatico, fecharTelaOrdemServico);
      this.habilitaSalvar(true);
      return false;
    }
    return true;
  }

  private validarSituacaoOSSubirVersao(automatico: boolean): boolean {
    if (
      !this.subirVersaoValido(automatico) &&
      !this.situacaoOSSubirVersaoInserir
    ) {
      this._notificationService.warning(
        NotificationConst.TITLE_WARNING,
        'Usuário sem permissão para utilizar situação!'
      );
      this.habilitaSalvar(true);
      return false;
    }
    return true;
  }

  private subirVersaoValido(automatico: boolean): boolean {
    let situacaoOSSubirVersao: Parametro =
      this.parametro.find(
        (parametro) =>
          parametro.nomeParametro ==
          ParametroConst.PARAMETRO_SITUACAO_OS_SUBIR_VERSAO
      ) || new Parametro();
    return (
      automatico ||
      situacaoOSSubirVersao.valorParametro !=
        this.formularioOs.controls['cmbSitAtual'].value.toString() ||
      this.modalCadastroListagemOrdemServicoTestado.jaFoiAberta
    );
  }

  private validarSituacaoOSInformacoesTrac(automatico: boolean): boolean {
    let situacao = this._situacaoAtual.find(
      (situacao) =>
        situacao.codsit == this.formularioOs.controls['cmbSitAtual'].value
    );

    if (
      automatico &&
      this.formularioOs.controls['cmbSitAtual'].value != '' &&
      situacao.integtrac == 'S' &&
      this.ordemServico.suptec.codticket == null
    ) {
      this.habilitaSalvar(true);
      return false;
    }
    return true;
  }

  private motivoOSNaoAceitaValido(automatico: boolean): boolean {
    let parametro = this.parametro.find(
      (parametro) =>
        parametro.nomeParametro ==
        ParametroConst.PARAMETRO_SITUACAO_OS_SOLICITA_MOTIVO_NAO_ACEITA
    );
    let situacaoOSMotivoNaoAceita = parametro ? parametro.valorParametro : '';
    return (
      automatico ||
      this.telaMotivoOSNaoAceitaValidar != null ||
      situacaoOSMotivoNaoAceita !=
        this.formularioOs.controls['cmbSitAtual'].value
    );
  }

  private validarCampos(validacao: String): boolean {
    switch (validacao) {
      case 'arrayValidacoes':
        if (this.formularioProduto.controls['cmbProduto'].value == null) {
          this._notificationService.warning(
            NotificationConst.TITLE_WARNING,
            'Preencha o campo Produto!'
          );
          this.cmbProduto.focus();
          return false;
        } else if (!this.formularioProduto.controls['cmbModulos'].value) {
          this._notificationService.warning(
            NotificationConst.TITLE_WARNING,
            'Preencha o campo Módulos!'
          );
          this.cmbModulos.focus();
          return false;
        } if (!this.formularioProduto.controls['cmbCategoria'].value) {
          this._notificationService.warning(
            NotificationConst.TITLE_WARNING,
            'Preencha o campo Categoria!'
          );
          this.cmbCategoria.focus();
          return false;
        } else if (!this.formularioProduto.controls['cmbTecnicos'].value) {
          this._notificationService.warning(
            NotificationConst.TITLE_WARNING,
            'Preencha o campo Técnico!'
          );
          this.cmbTecnicos.focus();
          return false;
        } else if(this.formularioOs.get('cmbSitAtual')?.invalid){
          this._notificationService.warning(
            NotificationConst.TITLE_WARNING,
            'Preencha o campo Situação!'
           )
          this.cmbSitAtual.focus();
          return false
        }else if (this.formularioOs.controls['cmbTipoAtendimento'].value == '') {
          this._notificationService.warning(
            NotificationConst.TITLE_WARNING,
            'Preencha o campo Atendimento!'
          );
          this.cmbTipoAtendimento.focus();
          return false;
        } else if (this.txtDescricaoServico == '') {
          this._notificationService.warning(
            NotificationConst.TITLE_WARNING,
            'Preencha o campo descrição dos serviços!'
          );
          this.txtDescricaoHTML.nativeElement.style.borderColor = 'red';
          this.txtDescricaoHTML.nativeElement.focus();
          return false;
        }
        return true;
      case 'arrayValidacoesCategoria':
        if (this.formularioProduto.controls['cmbCategoria'].value == 0) {
          this._notificationService.warning(
            NotificationConst.TITLE_WARNING,
            'Preencha o campo Categoria!'
          );
          this.cmbCategoria.focus();
          return false;
        }
        return true;
      case 'regExpTxtSolucao':
        if (this.txtSolucao == '') {
          this._notificationService.warning(
            NotificationConst.TITLE_WARNING,
            'Preencha o campo solução!'
          );
          this.txtSolucaoHTML.nativeElement.style.borderColor = 'red';
          this.txtSolucaoHTML.nativeElement.focus();
          return false;
        }
        return true;
      default:
        return true;
    }
  }

  private abrirTelaAgregado(
    apenasVisualizar: boolean,
    automatico: boolean,
    fecharTelaOrdemServico: boolean,
    imprimirEmSeguida: boolean,
    eventoF3: boolean
  ): boolean {
    if (!this.abrirTelaMotivoOSNaoAceita(automatico, fecharTelaOrdemServico)) {
      return false;
    }
    if (!this.abrirTelaRetornoProgramado(automatico, fecharTelaOrdemServico)) {
      return false;
    }
    if (
      !this.abrirTelaMotivoResultadoAlteracaoOS(
        apenasVisualizar,
        automatico,
        imprimirEmSeguida,
        fecharTelaOrdemServico
      )
    ) {
      return false;
    }
    if (
      !this.abrirTelaSubirVersao(
        automatico,
        imprimirEmSeguida,
        fecharTelaOrdemServico
      )
    ) {
      return false;
    }
    if (
      !this.abrirTelaInformacoesTrac(automatico, eventoF3, imprimirEmSeguida)
    ) {
      return false;
    }
    return true;
  }

  private abrirTelaMotivoOSNaoAceita(
    automatico: boolean,
    fecharTelaOrdemServico: boolean
  ): boolean {
    if (this.motivoOSNaoAceitaValido(automatico)) {
      return true;
    }
    this.telaMotivoOSNaoAceita.ordemServico = this.ordemServico;
    this.telaMotivoOSNaoAceita.fecharTelaOrdemServico = fecharTelaOrdemServico;
    this.telaMotivoOSNaoAceita.telaAnterior = this;
    this.telaMotivoOSNaoAceitaValidar = null;
    this.telaMotivoOSNaoAceita.open();

    this.habilitaSalvar(true);

    return false;
  }

  private abrirTelaSla(
    automatico: boolean,
    fecharTelaOrdemServico: boolean
  ): boolean {
    if (this.slaValido(automatico)) {
      return true;
    }
    if (!this.validarComboboxCategoria()) {
      return false;
    }

    this.abrirTelaSlaOS(automatico, fecharTelaOrdemServico);
    this.habilitaSalvar(true);

    return false;
  }

  private abrirTelaRetornoProgramado(
    automatico: boolean,
    fecharTelaOrdemServico: boolean
  ): boolean {
    if (this.retornoProgramadoValido(automatico)) {
      return true;
    }
    this.modalDetalheRetornoProgramado.ticketHistorico =
      this.ordemServico.ultimoHistorico;
    this.modalDetalheRetornoProgramado.ordemServico = this;
    this.modalDetalheRetornoProgramado.open();

    this.habilitaSalvar(true);
    this.isModalOpen = true;
    return false;
  }

  private retornoProgramadoValido(automatico: boolean): boolean {
    return (
      automatico ||
      this.formularioOs.controls['cmbTipoAtendimento'].value !=
        TipoAtendimento.VALOR_RETORNOPROGRAMADO ||
      (this.ordemServico.ultimoHistorico.dataRetornoProgramado != null &&
        this.ordemServico.ultimoHistorico.horaRetornoProgramado != '')
    );
  }

  private abrirTelaSubirVersao(
    automatico: boolean,
    imprimirEmSeguida: boolean,
    fecharTelaOrdemServico: boolean
  ): boolean {
    if (this.subirVersaoValido(automatico)) {
      return true;
    }

    this.modalCadastroListagemOrdemServicoTestado.irParaAbaSubirVersao =
      this.ordemServicoTestada;
    this.modalCadastroListagemOrdemServicoTestado.imprimirOSEmSeguida =
      imprimirEmSeguida;
    this.modalCadastroListagemOrdemServicoTestado.ticketPossuiRevisao =
      this.ticketPossuiRevisao;
    this.modalCadastroListagemOrdemServicoTestado.fecharTelaOrdemServico =
      fecharTelaOrdemServico;
    this.modalCadastroListagemOrdemServicoTestado.ticketFoiTestado =
      this.ordemServicoTestada;
    this.modalCadastroListagemOrdemServicoTestado.idOrdemServico =
      this.ordemServico.id.toString();
    this.modalCadastroListagemOrdemServicoTestado.telaAnterior = this;

    if (this.ordemServico.suptec.codsit != 0) {
      this.modalCadastroListagemOrdemServicoTestado.apenasVisualizar = true;
    }

    this.modalCadastroListagemOrdemServicoTestado.listarTicketTestado();

    this.modalCadastroListagemOrdemServicoTestado.open((testado: any) =>
      this.tratarResultadoTeste(testado)
    );

    this.habilitaSalvar(true);

    return false;
  }

  private abrirTelaInformacoesTrac(
    automatico: boolean,
    eventoF3: boolean,
    imprimirEmSeguida: boolean
  ): boolean {
    if (this.informacoesTracValido()) {
      return true;
    }

    let produto = this._produtos.find(
      (produto) =>
        produto.codigo == this.formularioProduto.controls['cmbProduto'].value
    );

    this.telaOrdemServicoInformacaoTrac.ordemServicoComponent = this;
    this.telaOrdemServicoInformacaoTrac.ordemServico = this.ordemServico;
    this.telaOrdemServicoInformacaoTrac.eventoF3 = eventoF3;
    this.telaOrdemServicoInformacaoTrac.produto = produto.descricao;
    this.telaOrdemServicoInformacaoTrac.imprimirEmSeguida = imprimirEmSeguida;
    this.telaOrdemServicoInformacaoTrac.open();
    this.habilitaSalvar(true);

    return false;
  }

  private informacoesTracValido(): boolean {
    let situacao = this._situacaoAtual.find(
      (situacao) =>
        situacao.codsit == this.formularioOs.controls['cmbSitAtual'].value
    );
    return (
      this.formularioOs.controls['cmbSitAtual'].value == null ||
      situacao.integtrac != 'S' ||
      this.ordemServico.suptec.codticket != null
    );
  }

  private async abrirTelaReabrirRetorno() {
    if (this.reabrirRetornoValido()) {
      return true;
    }

    await this._alertService.questionYesNo(
      NotificationConst.TITLE_ATENCAO,
      'Deseja reabrir a O.S. #' +
        this.ordemServico.suptec.codordorig +
        ' que gerou o retorno?',
      (this.ordemServico.reabrirOsQueGerouRetorno = 'S'),
      (this.ordemServico.reabrirOsQueGerouRetorno = 'N')
    );

    this.ordemServico.protec.usuario = this._usuario.login;
    this.salvar(false);

    return false;
  }

  private reabrirRetornoValido(): boolean {
    return (
      !this.reabrirOS ||
      this.ordemServico.suptec.codordorig == 0 ||
      (this.ordemServico.reabrirOsQueGerouRetorno != null &&
        this.ordemServico.reabrirOsQueGerouRetorno != '')
    );
  }

  private formatarTexto70caracteres(texto: string): string {
    const linhas = texto.split('\n');
    const linhasFormatadas = linhas.map((linha) => {
      let linhaFormatada = '';
      for (let i = 0; i < linha.length; i += 70) {
        linhaFormatada += linha.slice(i, i + 70) + '\n';
      }
      return linhaFormatada.trim();
    });
    return linhasFormatadas.join('\n');
  }

  async salvar(
    automatico: boolean = false,
    imprimirEmSeguida: boolean = false,
    eventoF3: boolean = false,
    logDeOndeEstaSalvando: String = 'Não definido'
  ) {
    try {
      this.ordemServico.usuarioLogado = this._usuario.id;
      let fecharTelaOrdemServico =
        this.fecharTelaOrdemServicoAposSalvarTelaFilha(automatico, eventoF3);
      let isSubirVersao = true;
      if (!this.validarAgregado(automatico, eventoF3, fecharTelaOrdemServico)) {
        return;
      }

      if (
        !this.abrirTelaAgregado(
          this.apenasVisualizar,
          automatico,
          fecharTelaOrdemServico,
          imprimirEmSeguida,
          eventoF3
        )
      ) {
        return;
      }

      await this.validacaoSubirVersao().then((res) => {
        isSubirVersao = res;
      });

      if (!isSubirVersao) {
        return;
      }

      if (this.atualizarProtec(this.ordemServico)) {
        this.ordemServico.atualizaProtec = 'S';
      }

      if (
        this.ordemServico.codigoTecnico.toString() !=
          this.formularioProduto.controls['cmbTecnicos'].value ||
        this.ordemServico.ultimoHistorico.tipoAtendimento !=
          this.formularioOs.controls['cmbTipoAtendimento'].value ||
        this.ordemServico.ultimoHistorico.solicitaLigacao
      ) {
        this.ordemServico.atualizaTicketHistorico = 'S';
      }

      if (
        this.ordemServico.suptec.codcat !=
        Number(this.formularioProduto.controls['cmbCategoria'].value)
      ) {
        this.ordemServico.usuarioLogado = this._usuario.id;
        this.ordemServico.atualizaCategoriaHistorico = 'S';
      }

      this.ordemServico.suptec.codord = this.ordemServico.id;
      this.ordemServico.suptec.codprodsg = Number(
        this.formularioProduto.controls['cmbProduto'].value
      );
      this.ordemServico.suptec.codcat = Number(
        this.formularioProduto.controls['cmbCategoria'].value
      );
      this.ordemServico.suptec.codsitant = Number(
        this.formularioOs.controls['cmbSitAtual'].value
      );
      this.ordemServico.suptec.descricaoOS = this.formatarTexto70caracteres(
        this.txtDescricaoServico.toString()
      );
      this.ordemServico.suptec.observacao = this.txtObservacao.toString();
      this.ordemServico.suptec.data =
        this.formularioOs.controls['txtData'].value;
      this.ordemServico.suptec.hora =
        this.formularioOs.controls['txtHorario'].value;

      let situacao = this._situacaoAtual.find(
        (situacao) =>
          situacao.codsit == this.formularioOs.controls['cmbSitAtual'].value
      );

      this.ordemServico.suptec.outroProc = 'N';
      if (situacao.outro_proc) {
        this.ordemServico.suptec.outroProc = situacao.outro_proc;
      }

      this.ordemServico.suptec.solucaoOS = this.formatarTexto70caracteres(
        this.txtSolucao.toString()
      );
      if (this.reabrirOS) {
        this.ordemServico.suptec.solucaoAntigaOS =
          this.ordemServico.suptec.solucaoOS;
        this.ordemServico.suptec.solucao1 = '';
        this.ordemServico.suptec.solucao2 = '';
        this.ordemServico.suptec.solucao3 = '';
        this.ordemServico.suptec.solucao4 = '';
        this.ordemServico.suptec.descricao4 = '';
        this.ordemServico.suptec.descricao5 = '';
        this.ordemServico.suptec.descricao11 = '';
        this.ordemServico.suptec.descricao12 = '';
        this.ordemServico.suptec.solucaoOS = '';
        this.ordemServico.reabrirOs = true;
      }

      this.ordemServico.suptec.modulatend =
        this.formularioProduto.controls['cmbModulos'].value;
      this.ordemServico.suptec.usufinal = '';
      this.ordemServico.suptec.codsit = 0;

      if (situacao.outro_proc != 'S') {
        this.ordemServico.suptec.usufinal = this._usuario.tabven.nome;
        this.ordemServico.suptec.codsit = this.ordemServico.suptec.codsitant;
      }

      this.ordemServico.protec.codsit = this.ordemServico.suptec.codsitant;

      this.ordemServico.protec.codord = this.ordemServico.id;
      this.ordemServico.protec.tecnico = Number(
        this.formularioProduto.controls['cmbTecnicos'].value
      );
      this.ordemServico.protec.codcat = Number(
        this.formularioProduto.controls['cmbCategoria'].value
      );
      this.ordemServico.contatoParceiro = this.ckParceiro ? 'S' : 'N';

      // Atualiza último TicketHistorico relacionado com o ticket que está sendo salvo
      let ticketHistorico: TicketHistorico = new TicketHistorico();
      ticketHistorico.fkIdCodord = this.ordemServico.suptec.codord;
      ticketHistorico.id =
        this.ordemServico.ultimoHistorico == null
          ? 0
          : this.ordemServico.ultimoHistorico.id;
      ticketHistorico.tipoAtendimentoAntigo =
        this.ordemServico.ultimoHistorico.tipoAtendimento;
      ticketHistorico.tipoAtendimento =
        this.formularioOs.controls['cmbTipoAtendimento'].value;
      ticketHistorico.solicitaLigacao =
        this.ordemServico.ultimoHistorico.solicitaLigacao;
      ticketHistorico.dataRetornoProgramado = '';
      ticketHistorico.horaRetornoProgramado = '';
      ticketHistorico.observacao = '';

      if (
        !automatico &&
        !eventoF3 &&
        this.formularioOs.controls['cmbTipoAtendimento'].value ==
          TipoAtendimento.VALOR_RETORNOPROGRAMADO
      ) {
        ticketHistorico.dataRetornoProgramado =
          this.ordemServico.ultimoHistorico.dataRetornoProgramado;
        ticketHistorico.horaRetornoProgramado =
          this.ordemServico.ultimoHistorico.horaRetornoProgramado;
        ticketHistorico.observacao =
          this.ordemServico.ultimoHistorico.observacao;
        this.ordemServico.atualizaTicketHistorico = 'S';
      }

      //Nao alterar o codigo do tecnico antes disso pois e necessario para pegar o tecnico antigo
      ticketHistorico.fkIdVendedorAntigo = this.ordemServico.codigoTecnico;
      ticketHistorico.fkIdVendedor = Number(
        this.formularioProduto.controls['cmbTecnicos'].value
      );
      ticketHistorico.fkIdUsuario = this._usuario.id;

      let tecnico = this._tecnicos.find(
        (tecnico) =>
          tecnico.codigo == this.formularioProduto.controls['cmbTecnicos'].value
      );

      this.ordemServico.codigoTecnico =
        this.formularioProduto.controls['cmbTecnicos'].value;
      this.ordemServico.nomeTecnico = tecnico.nome;
      this.ordemServico.detalhesAtendimento = this.txtDetalhesDoAtendimento
        ? this.txtDetalhesDoAtendimento.toString()
        : '';
      this.ordemServico.detalhesCliente = this.txtDetalhesDoCliente
        ? this.txtDetalhesDoCliente.toString()
        : '';

      this.atribuirModoSalvar(automatico, eventoF3, logDeOndeEstaSalvando);

      if (this.alterouSLA && this.slaValido(automatico)) {
        this.ordemServico.sla =
          this.formularioOs.controls['txtPrioridadeSLA'].value.toString();
        this.alterouSLA = false;
      }

      this.ordemServico.ultimoHistorico = ticketHistorico;

      this.ordemServico.suptec.integraComTrac = this.ordemServico.suptec
        .codticket
        ? 'S'
        : 'N';

      this.codCatAnterior = this.ordemServico.suptec.codcat;
      this.salvarNotificacao();

      //Inicio do tratamento de agendamentos
      let idDepartamentoSuporte = 0;
      if (RedirectRelatorio.getEmpresa() == 'rjk') {
        idDepartamentoSuporte = 1;
      } else {
        idDepartamentoSuporte = 2;
      }

      let cmbTipoAtendimento =
        this._tipoDeAtendimento.find(
          (atendimento) =>
            atendimento.value ==
            this.formularioOs.controls['cmbTipoAtendimento'].value
        ) || new TipoDeAtendimento();

      if (
        cmbTipoAtendimento.label != TipoAtendimento.LABEL_NENHUM &&
        this._usuario.departamento.id == idDepartamentoSuporte
      ) {
        if (
          this.historicoTecnico !=
          this.formularioProduto.controls['cmbTecnicos'].value
        ) {
          if (this.historioAtendimento != cmbTipoAtendimento.label) {
            if (
              cmbTipoAtendimento.label != TipoAtendimento.LABEL_RETORNO &&
              cmbTipoAtendimento.label !=
                TipoAtendimento.LABEL_RETORNOPROGRAMADO &&
              cmbTipoAtendimento.label != TipoAtendimento.LABEL_FINALIZADO
            ) {
              if (
                this.historioAtendimento == TipoAtendimento.LABEL_RETORNO ||
                this.historioAtendimento ==
                  TipoAtendimento.LABEL_RETORNOPROGRAMADO ||
                this.historioAtendimento == TipoAtendimento.LABEL_FINALIZADO
              ) {
                this.agendarTecnico(this.ordemServico.id, tecnico.codigo);
              } else {
                this.abrirAgendamentoNovo(tecnico.codigo);
              }
            } else {
              this.buscaAgendamentoOS();
            }
          } else {
            this.abrirAgendamentoNovo(tecnico.codigo);
          }
        } else {
          if (this.historioAtendimento != cmbTipoAtendimento.label) {
            if (
              cmbTipoAtendimento.label == TipoAtendimento.LABEL_RETORNO ||
              cmbTipoAtendimento.label ==
                TipoAtendimento.LABEL_RETORNOPROGRAMADO ||
              cmbTipoAtendimento.label == TipoAtendimento.LABEL_FINALIZADO
            ) {
              this.buscaAgendamentoOS();
            } else {
              if (
                this.historioAtendimento == TipoAtendimento.LABEL_RETORNO ||
                this.historioAtendimento ==
                  TipoAtendimento.LABEL_RETORNOPROGRAMADO ||
                this.historioAtendimento == TipoAtendimento.LABEL_FINALIZADO
              ) {
                this.agendarTecnico(this.ordemServico.id, tecnico.codigo);
              } else {
                await this.editarAgendamento();
              }
            }
          }
        }
      }
      //Fim do tratamento de agendamentos

      if (this._usuario.fkIdVendedor != tecnico.codigo) {
        this.notificacaoGlobo = new NotificacaoGlobo();
        this.notificacaoGlobo.fkIdTecnicoOrigem = this._usuario.fkIdVendedor;
        this.notificacaoGlobo.fkIdTecnicoDestino = tecnico.codigo;
        this.notificacaoGlobo.fkIdOrdemServico = this.ordemServico.id;

        this.salvarNotificacaoGlobo();
      }

      try {
        let res = await this._ordemservicoService.salvar(this.ordemServico);
        await this.onResultSalvar(res, automatico, imprimirEmSeguida, eventoF3);
      } catch (error: any) {
        console.error(error);
        this._notificationService.error(
          NotificationConst.TITLE_DEFAULT,
          error.message
        );
      }
    } catch (error: any) {
      console.error(error);
      this._notificationService.error(
        NotificationConst.TITLE_DEFAULT,
        error.message
      );
    }
  }

  private onResultSalvar(
    res: any,
    automatico: boolean,
    imprimirEmSeguida: boolean,
    eventoF3: boolean
  ) {
    if (automatico) {
      this.onResultSalvarAutomatico(res);
    } else if (imprimirEmSeguida) {
      this.imprimirOS();
    } else if (eventoF3) {
      this.onResultSalvarEventoF3(res);
    } else {
      this._notificationService.success(
        NotificationConst.TITLE_SUCCESS,
        'Ordem de serviço salva com sucesso'
      );
      this.onClose();
    }
    this.tituloOS = 'O.S:';
    this.reabrirOS = false;
  }

  private onResultSalvarEventoF3(res: any) {
    this.onResultSalvarAutomatico(res);
  }

  private onResultSalvarAutomatico(res: any) {
    this.atualizaCamposTela = true;
    this.salvouAutomatico = true;
    this.salvarClienteServidor();
    this.alterarRegistro(this.ordemServico.suptec.codord.toString());

    if (this.continuarOSAposSalvar) {
      this.continuarOSAposSalvar = false;
      this.abrirOSContinuacao();
    }

    let situacao = this._situacaoAtual.find(
      (situacao) =>
        situacao.codsit == this.formularioOs.controls['cmbSitAtual'].value
    );

    if (AmfUtil.validarResponse(res)) {
      if (
        situacao.outro_proc != 'S' &&
        !this.validarCampos('regExpTxtSolucao')
      ) {
        this.habilitaSalvar(true);
        return;
      }

      // Verifica se é uma O.S. de retorno e pergunta ao usuário se deseja finalizar a O.S. anterior.
      if (
        situacao.outro_proc != 'S' &&
        (this.ordemServico.suptec.codordorig != 0 ||
          this.ordemServico.retorno != 'N')
      ) {
        this._alertService.questionYesNo(
          'CONFIRMAÇÂO',
          'Esta O.S. possui retorno. Deseja fechá-las?',
          async () => {
            try {
              let res = await this._ordemservicoService.finalizaOSPaisEFilhas(
                this.ordemServico
              );
              if (AmfUtil.validarResponse(res)) {
                this._alertService.message(
                  NotificationConst.TITLE_ATENCAO,
                  res.data.body
                );
              }
            } catch (error: any) {
              console.error(error);
              this._notificationService.error(
                NotificationConst.TITLE_DEFAULT,
                error.message
              );
            }
          },
          () => this
        );
      }

      this._notificationService.success(
        NotificationConst.TITLE_SUCCESS,
        'Ordem de serviço salva com sucesso.'
      );
    }
    this.habilitaSalvar(true);
  }

  private async abrirOSContinuacao() {
    if (this.ordemServico.continuacao == 'N') {
      await this._alertService.questionYesNo(
        'Confirmação',
        'Deseja realmente abrir uma Ordem de Serviço de continuação?',
        async () => {
          this.habilitarHBoxContinuacao(false);
          try {
            let res = await this._ordemservicoService.localizar(
              this.ordemServico.suptec.codord
            );
            if (AmfUtil.validarResponse(res)) {
              this.onResultLocalizarParaAbrirOSContinuacao(res);
            }
          } catch (error: any) {
            console.error(error);
            this._notificationService.error(
              NotificationConst.TITLE_DEFAULT,
              error.message
            );
          }
        },
        () => this
      );
    } else {
      await this._alertService.message(
        NotificationConst.TITLE_ATENCAO,
        'Esta Ordem de Serviço já possui continuação: #' +
          this.ordemServico.continuacao
      );
    }
  }

  private async onResultLocalizarParaAbrirOSContinuacao(res: any) {
    this.ordemServicoAntiga = Object.assign(new OrdemServico(), res.data.body);

    this.ordemServicoAntiga.id = 0;
    this.ordemServicoAntiga.usuarioLogado = this._usuario.id;
    this.ordemServicoAntiga.suptec.codcli =
      this.ordemServicoAntiga.cliente.cadcli.codcli10;
    this.ordemServicoAntiga.suptec.usuaber = this._usuario.login
      .toUpperCase()
      .substring(0, 10);
    this.ordemServicoAntiga.tipoAtendimento = TipoAtendimento.VALOR_NENHUM;
    this.ordemServicoAntiga.suptec.integraComTrac =
      this.ordemServicoAntiga.suptec.codticket != null &&
      this.ordemServicoAntiga.suptec.codticket.length > 0
        ? 'S'
        : '';
    this.ordemServicoAntiga.suptec.codticket = '';
    this.ordemServicoAntiga.suptec.codordcont =
      this.ordemServicoAntiga.suptec.codord;
    this.ordemServicoAntiga.suptec.codordorig = 0;
    this.ordemServicoAntiga.suptec.osproxreto = 0;
    this.ordemServicoAntiga.suptec.osproxcont = 0;
    this.ordemServicoAntiga.suptec.codcat = Number(
      this.ordemServicoAntiga.suptec.codcat
    );
    this.ordemServicoAntiga.suptec.codsitant = 9;

    if (this.formularioProduto.controls['cmbProduto'].value != '') {
      this.ordemServicoAntiga.suptec.codprodsg =
        this.formularioProduto.controls['cmbProduto'].value;
    } else if (this.formularioProduto.controls['cmbCategoria'].value != '') {
      this.ordemServicoAntiga.suptec.codcat =
        this.formularioProduto.controls['cmbCategoria'].value;
    } else if (this.formularioProduto.controls['cmbModulos'].value != '') {
      this.ordemServicoAntiga.suptec.modulatend =
        this.formularioProduto.controls['cmbModulos'].value;
    }

    try {
      let res = await this._ordemservicoService.salvar(this.ordemServicoAntiga);
      if (AmfUtil.validarResponse(res)) {
        this.ordemServicoNovo = Object.assign(
          new OrdemServico(),
          res.data.body
        );
        this.alterarRegistro(this.ordemServicoNovo.suptec.codord.toString());
        if (!this.txtReabrirOSDisabled) {
          this.txtReabrirOS = this.ordemServicoNovo.suptec.codord.toString();
        }
        this.continuarOSAposSalvar = true;

        this.listarOrdPrev(this.ordemServico.id);
      }
    } catch (error: any) {
      console.error(error);
      this._notificationService.error(
        NotificationConst.TITLE_DEFAULT,
        error.message
      );
    }
  }

  private async salvarClienteServidor() {
    if (!this.clienteJaFoiCarregado) {
      return;
    }

    if (!this.carregouDadosClienteServidorLocal) {
      return;
    }

    this.clienteServidor.fkIdCliente = Number(
      this.ordemServico.cliente.cadcli.codcli10
    );
    if (
      this.formularioClienteServidor.controls['servidorCentralizado'].value !=
      ''
    ) {
      this.clienteServidor.servidorCentralizado =
        this.formularioClienteServidor.controls['servidorCentralizado'].value;
    } else {
      this.clienteServidor.servidorCentralizado = '';
    }
    this.clienteServidor.ipServidorExterno =
      this.formularioClienteServidor.controls['ipServidorExterno'].value;
    this.clienteServidor.ipServidorInterno =
      this.formularioClienteServidor.controls['ipServidorInterno'].value;
    this.clienteServidor.senhaRoot =
      this.formularioClienteServidor.controls['senhaRoot'].value;
    this.clienteServidor.usuarioServidor =
      this.formularioClienteServidor.controls['usuarioServidor'].value;
    this.clienteServidor.senhaServidor =
      this.formularioClienteServidor.controls['senhaServidor'].value;

    if (this.formularioClienteServidor.controls['tipoBancoDados'].value != '') {
      this.clienteServidor.tipoBancoDados =
        this.formularioClienteServidor.controls['tipoBancoDados'].value;
    } else {
      this.clienteServidor.tipoBancoDados = '';
    }
    this.clienteServidor.urlBanco =
      this.formularioClienteServidor.controls['urlBanco'].value;
    this.clienteServidor.portaBanco = Number(
      this.formularioClienteServidor.controls['portaBanco']
    );
    this.clienteServidor.usuarioBanco =
      this.formularioClienteServidor.controls['usuarioBanco'].value;
    this.clienteServidor.senhaBanco =
      this.formularioClienteServidor.controls['senhaBanco'].value;

    if (this.formularioClienteServidor.controls['utilizaTef'].value != '') {
      this.clienteServidor.utilizaTef =
        this.formularioClienteServidor.controls['utilizaTef'].value;
    } else {
      this.clienteServidor.utilizaTef = '';
    }
    this.clienteServidor.senhaVPN =
      this.formularioClienteServidor.controls['senhaVPN'].value;
    this.clienteServidor.nomeAdmRede =
      this.formularioClienteServidor.controls['nomeAdmRede'].value;
    this.clienteServidor.telefone =
      this.formularioClienteServidor.controls[
        'txtTelefoneContatoADMRede'
      ].value;
    this.clienteServidor.observacao =
      this.formularioClienteServidor.controls['txtObservacaoServidor'].value;

    try {
      let res = await this._clienteServidorService.salvar(this.clienteServidor);
      if (!AmfUtil.validarResponse(res)) {
        this._alertService.message(
          NotificationConst.TITLE_ATENCAO,
          'Falha ao salvar dados do Cliente Servidor'
        );
      }
    } catch (error: any) {
      console.error(error);
      this._notificationService.error(
        NotificationConst.TITLE_DEFAULT,
        error.message
      );
    }
  }

  private async salvarNotificacaoGlobo() {
    try {
      await this._notificacaoGloboService.novaNotificacao(
        this.notificacaoGlobo
      );
    } catch (error: any) {

  private async calcularPrevisaoOS() {
    let data = this.ordemServico.suptec.data;
    let hora: any[] = this.ordemServico.suptec.hora.toString().split(':');

    let dataOS: Date = new Date(DateField.dateFormat(data, 'DD/MM/YY'));
    dataOS.setHours(hora[0]);
    dataOS.setMinutes(hora[1]);

    let previsao: Date = new Date();
    previsao.setTime(dataOS.getTime() + this.prazoSLA * 60 * 60 * 1000);
    previsao.setHours(0);
    previsao.setMinutes(0);
    previsao.setSeconds(0);
    previsao.setMilliseconds(0);

    let ordprev: Ordprev = new Ordprev();
    ordprev.codord = this.ordemServico.id;
    ordprev.usuario = this._usuario.tabven.nome;
    ordprev.dataversao = DateField.dateToString(previsao, 'DD/MM/YYYY') || '';
    ordprev.justificat = '';
    ordprev.fkTipPrazo = this.idTipoPrazoPrevisao;

    try {
      await this._ordprevService.salvar(ordprev);
    } catch (error: any) {
      console.error(error);
      this._notificationService.error(
        NotificationConst.TITLE_DEFAULT,
        error.message
      );
    }
  }

  abrirOSRetorno() {
    if (this.ordemServico.retorno != 'N') {
      this._alertService.message(
        NotificationConst.TITLE_ATENCAO,
        'Esta Ordem de Serviço já Possui Retorno!\nOrdem de Serviço de Retorno #' +
          this.ordemServico.retorno
      );
      return;
    }

    if (this.ticketPossuiRevisao && !this.ordemServicoTestada) {
      this._alertService.message(
        NotificationConst.TITLE_ATENCAO,
        'O.S com revisão no TRAC necessita marcar como Testado para poder abrir retorno!'
      );
      return;
    }

    if (this.ordemServico.suptec.codordorig > 0) {
      this._alertService.questionYesNo(
        NotificationConst.TITLE_ATENCAO,
        'Esta Ordem de Serviço já é retorno da ' +
          this.ordemServico.suptec.codordorig +
          '\n\nDeseja abrir retorno mesmo assim?',
        () => this.confirmaAbrirRetorno(),
        () => this
      );
    } else {
      this._alertService.questionYesNo(
        NotificationConst.TITLE_ATENCAO,
        'Deseja realmente abrir uma Ordem de Serviço de retorno?',
        () => this.confirmaAbrirRetorno(),
        () => this
      );

  private abrirTelaTestado(
    isCliqueOpcaoRodape: boolean,
    automatico: boolean,
    fecharTelaOrdemServico: boolean
  ): boolean {
    if (this.ordemServicoTestadoValido(isCliqueOpcaoRodape, automatico)) {
      return true;
    }

    this.modalCadastroListagemOrdemServicoTestado.ticketPossuiRevisao =
      this.ticketPossuiRevisao;
    this.modalCadastroListagemOrdemServicoTestado.fecharTelaOrdemServico =
      fecharTelaOrdemServico;
    this.modalCadastroListagemOrdemServicoTestado.ticketFoiTestado =
      this.ordemServicoTestada;
    this.modalCadastroListagemOrdemServicoTestado.idOrdemServico =
      this.ordemServico.id.toString();
    this.modalCadastroListagemOrdemServicoTestado.telaAnterior = this;

    if (this.ordemServico.suptec.codsit != 0) {
      this.modalCadastroListagemOrdemServicoTestado.apenasVisualizar = true;
    }

    this.modalCadastroListagemOrdemServicoTestado.open((testado: any) =>
      this.tratarResultadoTeste(testado)
    );
    this.modalCadastroListagemOrdemServicoTestado.listarTicketTestado();

    this.habilitaSalvar(true);
    this.isModalOpen = true;

    return false;
  }

    if (resultado.length == 0) {

    }

  private ordemServicoTestadoValido(
    isCliqueOpcaoRodape: boolean,
    automatico: boolean
  ): boolean {
    if (isCliqueOpcaoRodape) {
      return false;
    }

    let situacao = this._situacaoAtual.find(
      (situacao) =>
        situacao.codsit == this.formularioOs.controls['cmbSitAtual'].value
    );

    return (
      automatico ||
      situacao.outro_proc == 'S' ||
      !this.ticketExisteTrac ||
      (this.tipoTicket != Ticket.TIPO_TAREFA &&
        this.tipoTicket != Ticket.TIPO_NP &&

        this.tipoTicket != Ticket.TIPO_ERRO_PARADA &&
        this.tipoTicket != Ticket.TIPO_ERRO_RETORNO) ||
      !this.ticketPossuiRevisao ||
      this.ordemServicoTestada
    );
  }

  private motivoAlteracaoValido(automatico: boolean): boolean {
    let visibleMotivo = true;

    if (automatico) {
      return visibleMotivo;
    }
    let valorParametro =
      this.parametro.find(
        (parametro) => parametro.nomeParametro == 'categoriaOSSolicitarMotivo'
      ) || new Parametro();
    let valorParametroCategoria = valorParametro.valorParametro.split(',');
    let valorParametroSit =
      this.parametro.find(
        (parametro) => parametro.nomeParametro == 'situacaoOSSolicitarMotivo'
      ) || new Parametro();
    let valorParametroSituacao = valorParametroSit.valorParametro.split(',');

    valorParametroCategoria.forEach((parametro) => {
      valorParametroSituacao.forEach((parametro2) => {
        if (
          this.telaCadastroMotivoAlteracao == null &&
          parametro2 == this.formularioOs.controls['cmbSitAtual'].value &&
          parametro == this.formularioProduto.controls['cmbCategoria'].value
        ) {
          visibleMotivo = false;
        }
      });
    });
    return visibleMotivo;
  }

  onResultListarUsuario(res: any) {
    this._cmbUsuarioOption = res.data.body.map(
      (usuario: any) =>
        new Option(`${usuario.id} - ${usuario.nome}`, usuario.id)
    );
    this._cmbUsuario = res.data.body;

    let todosUsuarios: Array<any> = new Array<any>(res.data.body);
    let usuariosDoCombo: Array<any> = new Array<any>(res.data.body);

    todosUsuarios.forEach((usuario) => {
      this._gdOrdemServicoNotificacao.forEach((ordemServicoNotificacao) => {
        if (ordemServicoNotificacao.fkIdUsuario == usuario.id) {
          let index: number = usuariosDoCombo.push(usuario);
          if (index >= 0) {
            usuariosDoCombo.splice(index, 1);
          }
        }
      });
    });

  async adicionarUsuario() {
    if (!this.ordemServicoNotificacaoInserir) {
      this._alertService.message(
        NotificationConst.TITLE_ATENCAO,
        'Você não tem as Permissoes Necessárias para Adicionar Notificações a Usuários'
      );
      return;
    }

    //colocado aqui para evitar mostrar mensagem sem ter realmente escolhido
    if (!this.cmbUsuario) {
      return;
    }

    let usuario: Usuario = new Usuario();

    await this._cmbUsuario.forEach((element) => {
      if (element.id == this.cmbUsuario) {
        usuario = element;
      }
    });

    let ordemServicoNotificacao: OrdemServicoNotificacao =
      new OrdemServicoNotificacao();

    ordemServicoNotificacao.fkIdUsuario = Number(this.cmbUsuario);
    ordemServicoNotificacao.fkIdOrdemServico = this.ordemServico.id;
    ordemServicoNotificacao.usuario = usuario as Usuario;

    this._gdOrdemServicoNotificacao.push(ordemServicoNotificacao);

    this._gdOrdemServicoNotificacao = [...this._gdOrdemServicoNotificacao];

    // Recria o selectOption sem o usuário selecionado.
    const index = this._cmbUsuario.indexOf(usuario, 0);
    if (index > -1) {
      this._cmbUsuario.splice(index, 1);
    }

    this._cmbUsuarioOption = this._cmbUsuario.map(
      (usuario) => new Option(`${usuario.id} - ${usuario.nome}`, usuario.id)
    );
  }

  excluirUsuario(row: any) {
    const indexDois = this._gdOrdemServicoNotificacao.indexOf(row, 0);
    if (indexDois > -1) {
      this._gdOrdemServicoNotificacao.splice(indexDois, 1);
    }

    this._gdOrdemServicoNotificacao = [...this._gdOrdemServicoNotificacao];
    this.listarUsuario();
  }

        );

      // Produtos de cliente

      // Trabalha com servidor central
      this.atribuirTrabalhaComServidorCentral(cliente[2]);

      // servidor do cliente
      this.atribuirClienteServidor(cliente[3]);

  async listarOrdemServico(limite: number) {
    let filtros: Filtro[] = [];
    let status: string;
    this.loadingListOs = true;

    if (!this.chkTodasFiliais) {
      filtros.push(
        new Filtro(
          FiltroConst.ORDEM_SERVICO_CODIGO_CLIENTE,
          this.ordemServico.cliente.cadcli.codcli10.toString()
        )
      );
    }

    if (this.dtDataDe != '') {
      filtros.push(
        new Filtro(FiltroConst.ORDEM_SERVICO_DATA_DE, this.dtDataDe)
      );
    }

    if (this.dtDataAte != '') {
      filtros.push(
        new Filtro(FiltroConst.ORDEM_SERVICO_DATA_ATE, this.dtDataAte)
      );
    }

    if (this.cmbStatus == 'TODAS_ABERTAS') {
      status = 'null';
    } else if (this.cmbStatus == 'TODAS_FECHADAS') {
      status = 'not null';
    } else {
      status = '';
    }
    filtros.push(new Filtro(FiltroConst.ORDEM_SERVICO_STATUS, status));

    if (this.chkTodasFiliais) {
      filtros.push(
        new Filtro(
          FiltroConst.ORDEM_SERVICO_CODIGO_SGSCLI,
          this.ordemServico.cliente.sgscli.empresa.toString()
        )
      );
    }

    try {
      let res = await this._ordemservicoService.listar(filtros, limite);
      if (AmfUtil.validarResponse(res)) {
        this.gdOrdemServico = res.data.body;
      }
    } catch (error: any) {
      console.error('OrdemDeServico.listarOrdemServico', error);
      this._notificationService.error(
        NotificationConst.TITLE_DEFAULT,
        error.message
      );
    }
    this.loadingListOs = false;
  }

      filtros.push(filtroDataDe);
    }

    if (this.dtDataAtualizacaoAte != '') {
      filtroDataAte = new Filtro(
        FiltroConst.ATUALIZACAO_SISTEMA_DATA_ATUALIZACAO_ATE,
        this.dtDataAtualizacaoAte
      );
      filtros.push(filtroDataAte);
    }
    try {
      let res = await this._atualizacaoSistemaService.listarUltimasVersoes(
        filtros
      );
      if (AmfUtil.validarResponse(res)) {
        this.gdAtualizacoes = res.data.body;
      }
    } catch (error: any) {
      console.error('OrdemServicoComponent.listarAtualizacao', error);
      this._notificationService.error(
        NotificationConst.TITLE_DEFAULT,
        error.message
      );
    }
    this.loadingAtualizar = false;
  }

  private async listarDadosTrac() {
    try {
      let res = await this._ticketService.localizar(this.txtReabrirOS);
      if (AmfUtil.validarResponse(res)) {
        let ticket = new Ticket();
        ticket = Object.assign(new Ticket(), res.data.body);
        this.txtProdutoTrac = this.ordemServico.produto;
        this.txtVersaoTrac = ticket.version;
        this.txtTipoTrac = ticket.type;
        if (ticket.pontos != null) {
          this.txtPontoTrac = ticket.pontos.toString();
        }
        if (ticket.pesoPlanejadoTeste != null) {
          this.txtPesoPlanejadoTicket = ticket.pesoPlanejadoTeste.toString();
        }
        this.txtVersaoGit = ticket.versiongit;
        if (ticket.requisito != null) {
          this.txtDescricaoTrac = FunctionsUtil.converterXhtmlParaRtf(
            ticket.requisito
          ).toString();
        }
        if (ticket.tarefa != null) {
          this.txtTarefaTrac = FunctionsUtil.converterXhtmlParaRtf(
            ticket.tarefa
          ).toString();
        }
        if (ticket.releaseNotes != null) {
          this.txtReleaseNoteTrac = ticket.releaseNotes;
        }
      }
    } catch (error: any) {
      console.error(error);
      this._notificationService.error(
        NotificationConst.TITLE_DEFAULT,
        error.message
      );

  private async validacaoSubirVersao(): Promise<boolean> {
    if (
      this.ordemServico.suptec.codsitant == 83 &&
      (this.formularioOs.controls['cmbSitAtual'].value != 83 ||
        this.formularioProduto.controls['cmbTecnicos'].value !=
          this.ordemServico.codigoTecnico)
    ) {
      return await this.listarVersoes();
    }
    return true;
  }

  private async listarVersoes(): Promise<boolean> {
    try {
      let res = await this._subirVersaoOsService.listar(
        this.ordemServico.id.toString()
      );
      if (AmfUtil.validarResponse(res)) {
        let versoes = Object.assign(new Array(), res.data.body);
        if (
          versoes.find(
            (v: any) => v.status == 'Autorizado' || v.status == 'Negado'
          )
        ) {
          return true;
        }
      }
      this._notificationService.warning(
        NotificationConst.TITLE_ATENCAO,
        "A situação anterior 'Subir versão' requer um status 'autorizado' ou 'negado' na tela de 'Testados' -> 'Subir Versão', para conseguir fazer a movimentação de situação ou técnico."
      );
      this.abrirSubirVersao();
      return false;
    } catch (error: any) {
      console.error(error.message);
      this._notificationService.error(
        NotificationConst.TITLE_DEFAULT,
        error.message
      );
    }
    return false;
  }

  @HostListener('window:keydown', ['$event'])
  atalhos(event: KeyboardEvent) {
    const fullCardName = `${this.cardName}${this.txtReabrirOS}`;

    // Se o modal estiver aberto, o ESC vai fechar o modal primeiro
    if (this.isModalOpen && event.key === 'Escape') {
      event.preventDefault();
      this.closeModal();
      return;
    }

    if (
      this._cardManagerService.getLastCard() === fullCardName
    ) {
        if (event.ctrlKey && event.key === 's') {
          event.preventDefault();
          this.salvar(false);
        } else if (event.key === 'F3') {
          event.preventDefault();
          this.salvar(false);
        } else if (event.key === 'Escape') {
          event.preventDefault();
          event.stopImmediatePropagation();
          this.onClose();
        }
    }
  }

  formatarTexto(event: Event): void {
    const textarea = event.target as HTMLTextAreaElement;
    const valorOriginal = textarea.value;
    const posicaoCursor = textarea.selectionStart;

    const aplicarRegras = (texto: string) => {
      if (!texto) return '';

      // A. Remove os inválidos e passa para maiúsculo
      const textoLimpo = texto.toUpperCase().replace(OrdemServicoComponent.regexInvalidos, '');

      // B. Separa as linhas baseadas nos Enters que estão no texto
      const linhas = textoLimpo.split('\n');
      const linhasFormatadas: string[] = [];

      // Variável para carregar a "sobra" de uma linha para a linha seguinte
      let overflow = '';

      // C. Processa linha por linha aplicando o "efeito cascata"
      for (let i = 0; i < linhas.length; i++) {
        // Junta o que sobrou da linha de cima com o conteúdo da linha atual
        const linhaAtual = overflow + linhas[i];

        if (linhaAtual.length > 70) {
          // Mantém os 70 primeiros caracteres nesta linha
          linhasFormatadas.push(linhaAtual.substring(0, 70));
          // O resto vira "sobra" para ser empurrada para o início da linha de baixo
          overflow = linhaAtual.substring(70);
        } else {
          // Se coube tudo na linha, adiciona e zera a sobra
          linhasFormatadas.push(linhaAtual);
          overflow = '';
        }
      }

      if (overflow) {
        const pedacos = overflow.match(/.{1,70}/g);
        if (pedacos) {
          linhasFormatadas.push(...pedacos);
        }
      }

      return linhasFormatadas.join('\n');
    };

    // Aplica as regras no texto inteiro e na fatia antes do cursor
    const valorFormatado = aplicarRegras(valorOriginal);
    const textoAntesDoCursor = valorOriginal.substring(0, posicaoCursor);
    const novaPosicaoCursor = aplicarRegras(textoAntesDoCursor).length;

    // Atualiza o DOM e a variável imediatamente
    textarea.value = valorFormatado;
    this.txtDescricaoServico = valorFormatado;

    textarea.setSelectionRange(novaPosicaoCursor, novaPosicaoCursor);
  }
